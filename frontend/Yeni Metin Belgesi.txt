# =========================
# Serve sitemap.xml as proper XML
# =========================
<Files "sitemap.xml">
  ForceType application/xml
  Header set Content-Type "application/xml"
</Files>

# =========================
# Canonical redirects + React SPA fallback
# =========================
<IfModule mod_rewrite.c>
  RewriteEngine On

  # One-step canonical: force HTTPS and non-www
  RewriteCond %{HTTPS} !=on [OR]
  RewriteCond %{HTTP_HOST} ^www\.kudushilali\.org$ [NC]
  RewriteRule ^ https://kudushilali.org%{REQUEST_URI} [R=301,L]

# =========================
# React Router / SPA fallback
# =========================
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /
  # Serve real files/folders as-is
  RewriteCond %{REQUEST_FILENAME} -f [OR]
  RewriteCond %{REQUEST_FILENAME} -d
  RewriteRule ^ - [L]
  # Everything else -> index.html
  RewriteRule . /index.html [L]
</IfModule>

# =========================
# MIME types (ensure AVIF works)
# =========================
AddType image/avif .avif

# =========================
# Compression (Brotli > gzip)
# =========================
<IfModule mod_brotli.c>
  BrotliCompressionQuality 5
  AddOutputFilterByType BROTLI_COMPRESS text/css text/html text/plain text/xml
  AddOutputFilterByType BROTLI_COMPRESS application/javascript application/json application/xml
  AddOutputFilterByType BROTLI_COMPRESS image/svg+xml
</IfModule>
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/css text/html text/plain text/xml
  AddOutputFilterByType DEFLATE application/javascript application/json application/xml
  AddOutputFilterByType DEFLATE image/svg+xml
</IfModule>

# =========================
# Cache-Control headers
# =========================
<IfModule mod_headers.c>
  # HTML: no cache so deploys reflect immediately
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache, must-revalidate, max-age=0"
  </FilesMatch>
  # Hashed JS/CSS files (safe for long cache)
  <FilesMatch ".*\.[0-9a-f]{8,}\.(js|css)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>


  # Hashed build assets under /static (safe to cache for 1 year)
  SetEnvIfExpr "%{REQUEST_URI} =~ m#^/static/.*\.(?:css|js|webp|avif|png|jpe?g|svg)(?:\?.*)?$#" IS_STATIC_HASHED=1
  Header set Cache-Control "public, max-age=31536000, immutable" env=IS_STATIC_HASHED

  # Non-versioned media you manage manually in /assets (30 days)
  SetEnvIfExpr "%{REQUEST_URI} =~ m#^/assets/.*\.(?:png|jpe?g|webp|avif|gif|svg|ico|woff2?|ttf|eot)(?:\?.*)?$#" IS_MEDIA_ASSET=1
  Header set Cache-Control "public, max-age=2592000" env=IS_MEDIA_ASSET

  # Allow fonts cross-origin
  <FilesMatch "\.(?:woff2?|ttf|otf|eot)$">
    Header set Access-Control-Allow-Origin "*"
  </FilesMatch>
</IfModule>

# =========================
# Expires (legacy support)
# =========================
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/html               "access plus 0 seconds"
  ExpiresByType text/css                "access plus 1 year"
  ExpiresByType application/javascript  "access plus 1 year"
  ExpiresByType text/javascript         "access plus 1 year"

  ExpiresByType image/avif              "access plus 30 days"
  ExpiresByType image/webp              "access plus 30 days"
  ExpiresByType image/png               "access plus 30 days"
  ExpiresByType image/jpeg              "access plus 30 days"
  ExpiresByType image/gif               "access plus 30 days"
  ExpiresByType image/svg+xml           "access plus 30 days"
  ExpiresByType image/x-icon            "access plus 30 days"

  ExpiresByType font/woff2              "access plus 30 days"
  ExpiresByType font/woff               "access plus 30 days"
  ExpiresByType font/ttf                "access plus 30 days"
  ExpiresByType font/otf                "access plus 30 days"
  ExpiresByType application/vnd.ms-fontobject "access plus 30 days"
</IfModule>
